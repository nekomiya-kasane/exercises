cmake_minimum_required(VERSION 3.12)

project(PikaPika CXX)

set(CXX_STANDARD_REQUIRED 17)

# collect sources
file(GLOB PROJSRC "*.h" "*.cpp")

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG	  "${CMAKE_CURRENT_SOURCE_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE	  "${CMAKE_CURRENT_SOURCE_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO "${CMAKE_CURRENT_SOURCE_DIR}/bin")

set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_DEBUG	  ${CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_RELEASE	  ${CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO ${CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE})

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_DEBUG          ${CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG})
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_RELEASE	  ${CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE})
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_RELWITHDEBINFO ${CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE})

add_executable(${PROJECT_NAME})
set_target_properties(${PROJECT_NAME} PROPERTIES LINKER_LANGUAGE CXX)


message(STATUS "Current Directory: ${CMAKE_CURRENT_SOURCE_DIR}")

target_sources(${PROJECT_NAME} PRIVATE ${PROJSRC})

set(EASYX_DIR ${CMAKE_SOURCE_DIR}/easyx)
#set(EASYX_LIB ${EASYX_DIR}/lib/VC2015/x64/EasyXw.lib)
set(EASYX_LIB ${EASYX_DIR}/lib/VC2015/x64/EasyXa.lib)

# link easyx
target_include_directories(${PROJECT_NAME} PUBLIC ${EASYX_DIR}/include)

target_link_libraries(${PROJECT_NAME} PUBLIC ${EASYX_LIB})

file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/resources" DESTINATION "${CMAKE_CURRENT_SOURCE_DIR}/bin")
